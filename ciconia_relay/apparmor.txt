#include <tunables/global>

profile ciconia_relay flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # Bashio asks the Supervisor for settings over HTTP
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Options and the relay's data
  /data/** rw,

  # The start script, then the relay in its own profile
  /run.sh rix,
  /usr/bin/relay cx -> relay,

  profile relay flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Stopped by S6-Overlay
    signal (receive) peer=*_ciconia_relay,

    # Tracker ports, the panel, MQTT, DNS
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network netlink raw,

    /usr/bin/relay rm,

    # Database (SQLite with locks) and the TLS certificate it makes
    /data/ r,
    /data/** rwk,

    # Your own certificate for the TLS port
    /ssl/ r,
    /ssl/** r,

    # Name resolution, CA certificates, time zones
    /etc/hosts r,
    /etc/resolv.conf r,
    /etc/nsswitch.conf r,
    /etc/services r,
    /etc/protocols r,
    /etc/localtime r,
    /usr/share/zoneinfo/** r,
    /etc/ssl/** r,
    /etc/ca-certificates/** r,
    /usr/share/ca-certificates/** r,

    # Go runtime: CPU limits, huge pages, listen backlog, host name
    @{PROC}/sys/kernel/hostname r,
    @{PROC}/sys/net/core/somaxconn r,
    @{PROC}/@{pid}/cgroup r,
    @{PROC}/@{pid}/mountinfo r,
    @{PROC}/@{pid}/stat r,
    /sys/fs/cgroup/** r,
    /sys/kernel/mm/transparent_hugepage/hpage_pmd_size r,

    /dev/tty rw,
  }
}
